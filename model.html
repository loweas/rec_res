# Function to process data for a single year
process_year <- function(year_input) {
  data_year <- reservation_data %>% filter(year == year_input)
  
  unique_visits <- data_year %>%
    filter(!is.na(CheckinDate)) %>%
    distinct(id, parkname, CheckinDate) %>%
    group_by(id, parkname) %>%
    summarize(VisitCount = n(), .groups = "drop")
  
  all_combinations <- expand.grid(
    id = unique(data_year$id),
    parkname = unique(data_year$parkname),
    stringsAsFactors = FALSE
  )
  
  visit_data <- all_combinations %>%
    left_join(unique_visits, by = c("id", "parkname")) %>%
    mutate(VisitCount = ifelse(is.na(VisitCount), 0, VisitCount))
  
  person_info <- data_year %>%
    select(id, Permittee, zip, Zip, NoOfGuests, days_visited, year) %>%
    distinct()
  
  long_resdata_count <- left_join(visit_data, person_info, by = "id")
  
  long_resdata_count <- long_resdata_count %>%
    mutate(island_park = case_when(
      parkname == "Ainapo Cabin" ~ "Hawaii",
      parkname == "Hipalau" ~ "Kauai",
      parkname == "Kaluahaulu" ~ "Kauai",
      parkname == "Kamananui Trail" ~ "Oahu",
      parkname == "Kaunala Trail" ~ "Oahu",
      parkname == "Kawaikoi" ~ "Kauai",
      parkname == "Keanakolu Ranger & Bunkhouse Cabins" ~ "Hawaii",
      parkname == "Kuaokala Trail" ~ "Oahu",
      parkname == "Kulanaahane Trail" ~ "Oahu",
      parkname == "Kuliouou Trail" ~ "Oahu",
      parkname == "Lonomea" ~ "Kauai",
      parkname == "Maakua Ridge Trail" ~ "Oahu",
      parkname == "Manana Trail" ~ "Oahu",
      parkname == "Peacock Flats Campsite" ~ "Oahu",
      parkname == "Sugi Grove" ~ "Kauai",
      parkname == "Waialae Cabin Campsite" ~ "Kauai",
      parkname == "Waikoali" ~ "Kauai",
      parkname == "Waikolu" ~ "Molokai",
      parkname == "Waimano Trail" ~ "Oahu",
      parkname == "Waimanu Campsite" ~ "Hawaii",
      parkname == "Wiliwili" ~ "Kauai",
      parkname == "Wiliwilinui Ridge Trail" ~ "Oahu",
      TRUE ~ NA_character_
    ))
  
  long_resdata_count <- long_resdata_count[order(long_resdata_count$id, long_resdata_count$parkname), ]
  long_resdata_count$parkname_numeric <- as.integer(factor(long_resdata_count$parkname, levels = unique(long_resdata_count$parkname)))
  
  long_resdata_count <- long_resdata_count %>%
    distinct(id, parkname, .keep_all = TRUE)
  
  final_data <- long_resdata_count %>%
    left_join(
      intra_island_distance %>%
        select(zip, parkname, income, residents_distance_meters, residents_duration, island_live),
      by = c("zip", "parkname")
    )
  
  final_data <- final_data %>%
    group_by(id) %>%
    mutate(visited_another_island = ifelse(any(island_park != island_live & VisitCount > 0), 1, 0)) %>%
    ungroup()
  
  final_data$travel_distance_km <- (final_data$residents_distance_meters / 1000) * 2
  
  
  return(final_data)
}

MDVEC

We will run a clean MDCEV. The official run for each year is :

mdcev_mle <- mdcev(~ 1, data = data_model, model = "gamma", algorithm = "MLE", std_errors = "mvn", n_draws = 100, trunc_data = 1, print_iterations = TRUE)

  • ~ 1: each site gets its own appeal (a constant).

  • algorithm = “MLE”: the model picks the parameter values that best fit the observed trips.

  • std_errors = “mvn”: uncertainty is measured by assuming the estimates are roughly normally distributed.

scale_gamma <- function(mod, f) {
  gc <- grep("^gamma", colnames(mod$stan_fit$theta_tilde))
  mod$stan_fit$theta_tilde[, gc] <- mod$stan_fit$theta_tilde[, gc] * f

  mod
}

rmdcev_losses <- function(yr, scenario) {
 
  file <- paste0("data/tc_constant/mdcev_cf_v2_", yr, "_", scenario, ".rds")
  if (file.exists(file)) return(readRDS(file))                        
 
  # this year's data with the counterfactual price (campers who cannot afford it are dropped)
  dat   <- get(paste0("final_data_", yr))
  keep  <- paste(dat$id, dat$parkname) %in% paste(camp$id[camp$year == yr], camp$parkname[camp$year == yr])
  dat   <- dat[keep, ]
  cf    <- camp[camp$year == yr, ]
  dat$price <- cf[[scenario]][match(paste(dat$id, dat$parkname), paste(cf$id, cf$parkname))]
 
  # estimate
  data_model <- mdcev.data(dat, id.var = "id", alt.var = "parkname", choice = "VisitCount")
  fit <- mdcev(~ 1, data = data_model, model = "gamma", algorithm = "MLE", std_errors = "mvn",
               n_draws = N_DRAWS, trunc_data = 1, print_iterations = FALSE)
 
  # policies: close campsite k (its price is set very high), then close all campsites
  J <- fit$stan_data[["J"]]
  policies <- CreateBlankPolicies(npols = J + 1, model = fit, price_change_only = TRUE)
  policies$price_p[seq_len(J)] <- lapply(seq_len(J), function(k) { v <- policies$price_p[[k]]; v[k + 1] <- 999999; v })
  policies$price_p[[J + 1]]    <- c(0, rep(999999, J))
 
  # conditional welfare
  df_sim <- PrepareSimulationData(fit, policies)
  set.seed(2000 + yr)
  w <- mdcev.sim(df_sim$df_indiv, df_common = df_sim$df_common, sim_options = df_sim$sim_options,
                 cond_err = 1, nerrs = NERRS, draw_mlhs = 1, sim_type = "welfare")
 
  # mean over draws for each camper and policy; average loss over the single-campsite policies (the last policy closes everything)
  per_camper <- t(sapply(w, colMeans, na.rm = TRUE))                    # campers x policies
  out <- tibble(id = as.character(unique(dat$id)),
                loss = -rowMeans(per_camper[, seq_len(J), drop = FALSE]))
  saveRDS(out, file)
  out
}

## give each trip a week, at most one trip per week. pref = preferred (calendar) week, whose week is taken goes to the nearest free week (the earlier one on a tie). Undated trips go to the first free week.

assign_weeks <- function(pref) {
  n <- length(pref)
  if (n > 52) stop("a camper has more than 52 trips; cannot place one trip per week")
  out <- rep(NA_integer_, n); used <- logical(52)
  for (k in order(is.na(pref), pref)) {
    w <- if (is.na(pref[k])) 1L else as.integer(pref[k])
    if (used[w]) { free <- which(!used); w <- free[which.min(abs(free - w))] }
    out[k] <- w; used[w] <- TRUE
  }
  out
}

# MDVEC

We will run a clean MDCEV. The official run for each year is :

```
mdcev_mle <- mdcev(~ 1, data = data_model, model = "gamma", algorithm = "MLE", std_errors = "mvn", n_draws = 100, trunc_data = 1, print_iterations = TRUE)
```

-   \~ 1: each site gets its own appeal (a constant).

-   algorithm = "MLE": the model picks the parameter values that best fit the observed trips.

-   std_errors = "mvn": uncertainty is measured by assuming the estimates are roughly normally distributed.

-   n_draws = 100: it saves 100 plausible sets of parameters for confidence intervals on welfare.

-   trunc_data = 1: everyone in the data took at least one trip.

-   print_iterations = TRUE: just shows progress.

## Welfare

This section tales the results from the mdcev_model_list.rds and simulates welfare by increasing the price of of each price for a closure.

What this does slightly different than the sims.

Before we used

-   **`n_draws = 100`** whic saves 100 plausible sets of parameters.

-   We run a simulated **100-draw version** adds `nsims = 100`, so it uses all 100 and not the default which is == 30

-   This is stored in data/mdcev/welfare_100/

  print(x$CoefTable, digits = digits, n = Inf); invisible(x)
}
 
welfare_list_100 <- list()
for (yr in years) {
  y   <- as.character(yr)
  mod <- mdcev_model_list[[y]]
  nalts <- mod$stan_data$J
  message(">>> ", y, ": ", nalts, " sites, ", NSIMS, " parameter draws")
 
  ## policies: 1..nalts = close site i, nalts + 1 = close all sites
  policies <- CreateBlankPolicies(npols = nalts + 1, model = mod, price_change_only = TRUE)
  for (i in 1:nalts) policies$price_p[[i]][i + 1] <- 999999      # +1 skips the numeraire
  policies$price_p[[nalts + 1]] <- c(0, rep(999999, nalts))
  stopifnot(all(sapply(policies$price_p, length) == nalts + 1))
 
  set.seed(898989 + yr)
  df_sim <- PrepareSimulationData(mod, policies, nsims = NSIMS)
 
  t0 <- Sys.time()
  w <- mdcev.sim(df_sim$df_indiv,
                 df_common   = df_sim$df_common,
                 sim_options = df_sim$sim_options,
                 cond_err    = 1,
                 nerrs       = NERRS,
                 draw_mlhs   = 1,
                 sim_type    = "welfare")
  message("    done in ", format(Sys.time() - t0))
 
  stopifnot(nrow(as.matrix(w[[1]])) == NSIMS)                    # really 100 draws
  welfare_list_100[[y]] <- w
  saveRDS(w, sprintf("data/mdcev/welfare_%s.rds", y))
  print(summary(w))
}
saveRDS(welfare_list_100, "data/mdcev/welfare_list_100.rds")

    left_join(zips, by = "id")
})

## give each trip a week, at most one trip per week. pref = preferred (calendar) week, whose week is taken goes to the nearest free week (the earlier one on a tie). Undated trips go to the first free week.

assign_weeks <- function(pref) {
  n <- length(pref)
  if (n > 52) stop("a camper has more than 52 trips; cannot place one trip per week")
  out <- rep(NA_integer_, n); used <- logical(52)
  for (k in order(is.na(pref), pref)) {
    w <- if (is.na(pref[k])) 1L else as.integer(pref[k])
    if (used[w]) { free <- which(!used); w <- free[which.min(abs(free - w))] }
    out[k] <- w; used[w] <- TRUE
  }
  out
}
