calc_summary <- function(data, loss_col, years) {
    data %>%
        filter(year %in% years, parkname != "Stay_Home") %>%
        group_by(parkname, year) %>%
        summarise(
            n        = n(),
            mean_val = mean(.data[[loss_col]],  na.rm = TRUE),
            sd_val   = sd(.data[[loss_col]],    na.rm = TRUE),
            se_val   = sd_val / sqrt(n),
            LCI      = mean_val - 1.96 * se_val,
            UCI      = mean_val + 1.96 * se_val,
            .groups  = "drop"
        ) %>%
        select(parkname, year, mean_val, LCI, UCI)
}

nice_breaks <- function(lo, hi, n = 5) {
  steps <- c(1, 2, 2.5, 5) * 10^rep(-3:3, each = 4)
  for (s in steps) { b0 <- floor(lo / s) * s; if (b0 + (n - 1) * s >= hi) return(b0 + s * (0:(n - 1))) }
}

library(ggplot2)
years <- 2018:2023
incs  <- c(1, 10, 20, 50)
s_vec <- incs / 100

## ---- Repeated RUM: change in expected trips (52 x site probabilities), Krinsky-Robb CI ----
rum_el <- function(m, seed, R = 1000) {
  X    <- model.matrix(m)[grepl("_w1$", idx(m, 1)), ]       # week 1 only: all 52 weeks are identical
  alts <- unique(as.character(idx(m, 2))); site <- alts != "Stay_Home"
  trips <- function(b, s) {                                  # expected trips with costs x (1 + s)
    Xs <- X; Xs[, "price"] <- (1 + s) * Xs[, "price"]
    E  <- exp(matrix(Xs %*% b[colnames(X)], ncol = length(alts), byrow = TRUE))
    sum((E / rowSums(E))[, site])
  }
  el <- function(b) sapply(s_vec, function(s) (trips(b, s) / trips(b, 0) - 1) / s)
  set.seed(seed)
  kr <- t(apply(MASS::mvrnorm(R, coef(m), vcov(m)), 1, el))
  data.frame(est = el(coef(m)), lo = apply(kr, 2, quantile, .025), hi = apply(kr, 2, quantile, .975))
}

## ---- MDCEV: conditional demand, budget holds (numeraire solved by bisection) --------------
md_el <- function(mod) {
  sd <- mod$stan_data; x <- sd$quant_j; p <- sd$price_j; y <- as.numeric(sd$income)
  k  <- rowSums(x) > 0; x <- x[k, ]; p <- p[k, ]; y <- y[k]  # campers with trips
  z0 <- y - rowSums(p * x)
  el <- function(g, a1, s) {
    G   <- matrix(g, nrow(x), ncol(x), byrow = TRUE)
    dem <- function(z) pmax(0, (G + x) / (1 + s) * (z / z0)^(1 - a1) - G) * (x > 0)
    lo <- 0 * y; hi <- y
    for (i in 1:60) {                                        # numeraire z that exhausts income
      z <- (lo + hi) / 2; up <- y - z - (1 + s) * rowSums(p * dem(z)) > 0
      lo[up] <- z[up]; hi[!up] <- z[!up]
    }
    (sum(dem(z)) / sum(x) - 1) / s
  }
  th <- as.matrix(mod$stan_fit$theta_tilde)
  gc <- grep("^gamma", colnames(th)); ac <- grep("^alpha", colnames(th))[1]
  E  <- t(apply(th, 1, function(r) sapply(s_vec, function(s) el(r[gc], r[ac], s))))
  data.frame(est = colMeans(E), lo = apply(E, 2, quantile, .025), hi = apply(E, 2, quantile, .975))
}

  data.frame(model = "Repeated RUM", year = y, inc = incs,

rum_losses <- function(d) {
 
  ids   <- sort(unique(d$id))
  sites <- sort(unique(d$parkname))
  put   <- function(v) { m <- matrix(NA_real_, length(ids), length(sites))     # camper x campsite table
                         m[cbind(match(d$id, ids), match(d$parkname, sites))] <- v; m }
  price <- put(d$price)
  trips <- put(d$trips)
  miles <- put(d$miles)
  extra <- put(d$leftover)
  rate  <- put(d$rate_per_mile)[, 1]                                            # one rate per camper
 
  # --- estimate kappa (one per campsite) and lambda by maximum likelihood ---
  log_lik <- function(theta) {
    kappa <- theta[seq_along(sites)];  beta <- theta[length(sites) + 1]
    V     <- sweep(beta * price, 2, kappa, "+")                                  # utility of each campsite
    denom <- 1 + rowSums(exp(V))                                                 # 1 = stay home
    sum(trips * V) - 52 * sum(log(denom))
  }
  start <- c(log((colSums(trips) + 0.5) / (52 * length(ids))), -0.01)
  fit   <- optim(start, function(th) -log_lik(th), method = "BFGS", control = list(maxit = 2000, reltol = 1e-12))
  kappa <- fit$par[seq_along(sites)]
  beta  <- fit$par[length(sites) + 1]
  lambda <- -beta                                                               # marginal utility of income
 
  # --- loss per campsite closure, averaged over the campsites, for a given price table ---
  average_loss <- function(price_table) {
    V     <- sweep(beta * price_table, 2, kappa, "+")
    P     <- exp(V) / (1 + rowSums(exp(V)))                                      # weekly probability of each campsite
    rowMeans(-52 * log(1 - P) / lambda)
  }
 
  # --- four price scenarios ---
  avg_rate  <- mean(rate)                                                       # the average camper's rate per mile
  avg_miles <- colMeans(miles)                                                  # the average camper's distance to each campsite
 
  price_baseline       <- price                                                 # everything as observed
  price_equal_rate     <- avg_rate * miles + extra                              # same rate per mile, own distance  -> value of time OFF
  price_equal_distance <- outer(rate, avg_miles) + extra                        # own rate per mile, same distance  -> distance OFF
  price_both_equal     <- matrix(avg_rate * avg_miles, length(ids), length(sites), byrow = TRUE) + extra
 
  tibble(id = ids,
         baseline       = average_loss(price_baseline),
         equal_rate     = average_loss(pmax(price_equal_rate, 0)),
         equal_distance = average_loss(pmax(price_equal_distance, 0)),
         both_equal     = average_loss(pmax(price_both_equal, 0)))
}

rmdcev_losses <- function(yr, scenario) {

  file <- paste0("data/mdcev_cf_", yr, "_", scenario, ".rds")
  if (file.exists(file)) return(readRDS(file))                          # already done: reuse

  # this year's data with the counterfactual price (campers who cannot afford it are dropped)
  dat   <- get(paste0("final_data_", yr))
  keep  <- paste(dat$id, dat$parkname) %in% paste(camp$id[camp$year == yr], camp$parkname[camp$year == yr])
  dat   <- dat[keep, ]
  cf    <- camp[camp$year == yr, ]
  dat$price <- cf[[scenario]][match(paste(dat$id, dat$parkname), paste(cf$id, cf$parkname))]

  # estimate
  data_model <- mdcev.data(dat, id.var = "id", alt.var = "parkname", choice = "VisitCount")
  fit <- mdcev(~ 1, data = data_model, model = "gamma", algorithm = "MLE", std_errors = "mvn",
               n_draws = N_DRAWS, trunc_data = 1, print_iterations = FALSE)

  # policies: close campsite k (its price is set very high), then close all campsites
  J <- fit$stan_data[["J"]]
  policies <- CreateBlankPolicies(npols = J + 1, model = fit, price_change_only = TRUE)
  policies$price_p[seq_len(J)] <- lapply(seq_len(J), function(k) { v <- policies$price_p[[k]]; v[k + 1] <- 999999; v })
  policies$price_p[[J + 1]]    <- c(0, rep(999999, J))

  # conditional welfare
  df_sim <- PrepareSimulationData(fit, policies)
  set.seed(2000 + yr)
  w <- mdcev.sim(df_sim$df_indiv, df_common = df_sim$df_common, sim_options = df_sim$sim_options,
                 cond_err = 1, nerrs = NERRS, draw_mlhs = 1, sim_type = "welfare")

  # mean over draws for each camper and policy; average loss over the single-campsite policies (the last policy closes everything)
  per_camper <- t(sapply(w, colMeans, na.rm = TRUE))                    # campers x policies
  out <- tibble(id = as.character(unique(dat$id)),
                loss = -rowMeans(per_camper[, seq_len(J), drop = FALSE]))
  saveRDS(out, file)
  out
}

##For one year: the MDCEV loss under the four price tables ------------------------------------------
mdcev_losses <- function(yr) {

  fit <- mdcev_model_list[[as.character(yr)]]
  sd  <- fit$stan_data                                   # trips, prices and income in the order rmdcev uses
  x <- sd$quant_j;  p <- sd$price_j;  y <- as.numeric(sd$income);  N <- sd$I;  J <- sd$J

  sites <- as.character(fit$parms_info$alt_names)       # campsites in the model's order
  d     <- filter(camp, year == yr)

  # camper x campsite tables, in the same order as p. rmdcev keeps campers in the order of the data: try that order first
  ids   <- as.character(unique(final_data[final_data$year == yr, ]$id))
  put   <- function(v) { m <- matrix(NA_real_, N, J)
                         m[cbind(match(d$id, ids), match(d$parkname, sites))] <- v; m }
  if (max(abs(put(d$price) - p), na.rm = TRUE) > 1e-6) {  # otherwise try sorted ids
    ids <- sort(unique(as.character(d$id)))
  }
  price <- put(d$price)
  miles <- put(d$miles)
  extra <- put(d$leftover)
  rate  <- put(d$rate_per_mile)[, 1]
  stopifnot("price table does not match the model's price table" = max(abs(price - p), na.rm = TRUE) < 1e-6)

  # estimated parameters (point estimates), as in your fast script
  pt <- unlist(fit$stan_fit$par[c("psi", "gamma", "alpha", "scale")])
  names(pt) <- c(paste0("psi[1,", seq_len(J - 1), "]"), paste0("gamma[1,", seq_len(J), "]"), "alpha[1,1]", "scale[1]")
  lpsi  <- c(0, pt[grepl("^psi\\[", names(pt))])
  g     <- pt[grepl("^gamma\\[", names(pt))]
  a1    <- pt[grepl("^alpha\\[", names(pt))][1]
  scale <- pt[grepl("^scale\\[", names(pt))][1]

  # four price tables. The MDCEV needs every price to be positive (it divides by price), so a counterfactual price below $1
  # (possible when the leftover is negative at very short distances) is set to $1; the message says how many.
  floor_price <- function(m, label) {
    n <- sum(m < 1)
    if (n > 0) message("  ", yr, ", ", label, ": ", n, " of ", length(m), " prices below $1 were set to $1")
    pmax(m, 1)
  }
  avg_rate  <- mean(rate)
  avg_miles <- colMeans(miles)
  price_equal_rate     <- floor_price(avg_rate * miles + extra,                         "equal rate")
  price_equal_distance <- floor_price(outer(rate, avg_miles) + extra,                   "equal distance")
  price_both_equal     <- floor_price(matrix(avg_rate * avg_miles, N, J, byrow = TRUE) + extra, "both equal")

  # a camper must be able to afford the trips at the counterfactual prices (the model needs money left for the numeraire):
  # campers who would spend more than 95% of income on trips in ANY of the four price tables are left out of all four
  tables <- list(baseline = price, equal_rate = price_equal_rate, equal_distance = price_equal_distance, both_equal = price_both_equal)
  affordable <- Reduce(`&`, lapply(tables, function(tb) y - rowSums(tb * x) > 0.05 * y))
  message("  ", yr, ": ", sum(!affordable), " of ", N, " campers left out (trip spending above 95% of income in some price table)")

  # average loss per single-campsite closure, for a given price table (same draws for every scenario)
  average_loss <- function(price_table) {
    stopifnot("prices must be positive and not missing" = !anyNA(price_table[affordable, ]) && all(price_table[affordable, ] > 0))
    set.seed(1)
    xs <- x[affordable, , drop = FALSE];  ps <- price_table[affordable, , drop = FALSE];  ys <- y[affordable]
    CD <- md_cond_draws(xs, ps, ys, g, a1, lpsi, scale, NERRS)
    out <- rep(NA_real_, N)
    out[affordable] <- rowMeans(md_welfare_person_cond(CD, xs, ps, ys, g, a1))
    out
  }

  tibble(id = ids,
         baseline       = average_loss(price),
         equal_rate     = average_loss(price_equal_rate),
         equal_distance = average_loss(price_equal_distance),
         both_equal     = average_loss(price_both_equal))
}

## ---- Repeated RUM: change in expected trips (52 x site probabilities), Krinsky-Robb CI ----
rum_el <- function(m, seed, R = 1000) {
  X    <- model.matrix(m)[grepl("_w1$", idx(m, 1)), ]       # week 1 only: all 52 weeks are identical
  alts <- unique(as.character(idx(m, 2))); site <- alts != "Stay_Home"
  trips <- function(b, s) {                                  # expected trips with costs x (1 + s)
    Xs <- X; Xs[, "price"] <- (1 + s) * Xs[, "price"]
    E  <- exp(matrix(Xs %*% b[colnames(X)], ncol = length(alts), byrow = TRUE))
    sum((E / rowSums(E))[, site])
  }
  el <- function(b) sapply(s_vec, function(s) (trips(b, s) / trips(b, 0) - 1) / s)
  set.seed(seed)
  kr <- t(apply(MASS::mvrnorm(R, coef(m), vcov(m)), 1, el))
  data.frame(est = el(coef(m)), lo = apply(kr, 2, quantile, .025), hi = apply(kr, 2, quantile, .975))
}

## ---- MDCEV: conditional demand, budget holds (numeraire solved by bisection) --------------
md_el <- function(mod) {
  sd <- mod$stan_data; x <- sd$quant_j; p <- sd$price_j; y <- as.numeric(sd$income)
  k  <- rowSums(x) > 0; x <- x[k, ]; p <- p[k, ]; y <- y[k]  # campers with trips
  z0 <- y - rowSums(p * x)
  el <- function(g, a1, s) {
    G   <- matrix(g, nrow(x), ncol(x), byrow = TRUE)
    dem <- function(z) pmax(0, (G + x) / (1 + s) * (z / z0)^(1 - a1) - G) * (x > 0)
    lo <- 0 * y; hi <- y
    for (i in 1:60) {                                        # numeraire z that exhausts income
      z <- (lo + hi) / 2; up <- y - z - (1 + s) * rowSums(p * dem(z)) > 0
      lo[up] <- z[up]; hi[!up] <- z[!up]
    }
    (sum(dem(z)) / sum(x) - 1) / s
  }
  th <- as.matrix(mod$stan_fit$theta_tilde)
  gc <- grep("^gamma", colnames(th)); ac <- grep("^alpha", colnames(th))[1]
  E  <- t(apply(th, 1, function(r) sapply(s_vec, function(s) el(r[gc], r[ac], s))))
  data.frame(est = colMeans(E), lo = apply(E, 2, quantile, .025), hi = apply(E, 2, quantile, .975))
}

rum_losses <- function(d) {
 
  ids   <- sort(unique(d$id))
  sites <- sort(unique(d$parkname))
  put   <- function(v) { m <- matrix(NA_real_, length(ids), length(sites))     # camper x campsite table
                         m[cbind(match(d$id, ids), match(d$parkname, sites))] <- v; m }
  price <- put(d$price)
  trips <- put(d$trips)
  miles <- put(d$miles)
  extra <- put(d$leftover)
  rate  <- put(d$rate_per_mile)[, 1]                                            # one rate per camper
 
  # --- estimate kappa (one per campsite) and lambda by maximum likelihood ---
  log_lik <- function(theta) {
    kappa <- theta[seq_along(sites)];  beta <- theta[length(sites) + 1]
    V     <- sweep(beta * price, 2, kappa, "+")                                  # utility of each campsite
    denom <- 1 + rowSums(exp(V))                                                 # 1 = stay home
    sum(trips * V) - 52 * sum(log(denom))
  }
  start <- c(log((colSums(trips) + 0.5) / (52 * length(ids))), -0.01)
  fit   <- optim(start, function(th) -log_lik(th), method = "BFGS", control = list(maxit = 2000, reltol = 1e-12))
  kappa <- fit$par[seq_along(sites)]
  beta  <- fit$par[length(sites) + 1]
  lambda <- -beta                                                               # marginal utility of income
 
  # --- loss per campsite closure, averaged over the campsites, for a given price table ---
  average_loss <- function(price_table) {
    V     <- sweep(beta * price_table, 2, kappa, "+")
    P     <- exp(V) / (1 + rowSums(exp(V)))                                      # weekly probability of each campsite
    rowMeans(-52 * log(1 - P) / lambda)
  }
 
  # --- four price scenarios ---
  avg_rate  <- mean(rate)                                                       # the average camper's rate per mile
  avg_miles <- colMeans(miles)                                                  # the average camper's distance to each campsite
 
  price_baseline       <- price                                                 # everything as observed
  price_equal_rate     <- avg_rate * miles + extra                              # same rate per mile, own distance  -> value of time OFF
  price_equal_distance <- outer(rate, avg_miles) + extra                        # own rate per mile, same distance  -> distance OFF
  price_both_equal     <- matrix(avg_rate * avg_miles, length(ids), length(sites), byrow = TRUE) + extra
 
  tibble(id = ids,
         baseline       = average_loss(price_baseline),
         equal_rate     = average_loss(pmax(price_equal_rate, 0)),
         equal_distance = average_loss(pmax(price_equal_distance, 0)),
         both_equal     = average_loss(pmax(price_both_equal, 0)))
}

rmdcev_losses <- function(yr, scenario) {

  file <- paste0("data/mdcev_cf_", yr, "_", scenario, ".rds")
  if (file.exists(file)) return(readRDS(file))                          # already done: reuse

  # this year's data with the counterfactual price (campers who cannot afford it are dropped)
  dat   <- get(paste0("final_data_", yr))
  keep  <- paste(dat$id, dat$parkname) %in% paste(camp$id[camp$year == yr], camp$parkname[camp$year == yr])
  dat   <- dat[keep, ]
  cf    <- camp[camp$year == yr, ]
  dat$price <- cf[[scenario]][match(paste(dat$id, dat$parkname), paste(cf$id, cf$parkname))]

  # estimate
  data_model <- mdcev.data(dat, id.var = "id", alt.var = "parkname", choice = "VisitCount")
  fit <- mdcev(~ 1, data = data_model, model = "gamma", algorithm = "MLE", std_errors = "mvn",
               n_draws = N_DRAWS, trunc_data = 1, print_iterations = FALSE)

  # policies: close campsite k (its price is set very high), then close all campsites
  J <- fit$stan_data[["J"]]
  policies <- CreateBlankPolicies(npols = J + 1, model = fit, price_change_only = TRUE)
  policies$price_p[seq_len(J)] <- lapply(seq_len(J), function(k) { v <- policies$price_p[[k]]; v[k + 1] <- 999999; v })
  policies$price_p[[J + 1]]    <- c(0, rep(999999, J))

  # conditional welfare
  df_sim <- PrepareSimulationData(fit, policies)
  set.seed(2000 + yr)
  w <- mdcev.sim(df_sim$df_indiv, df_common = df_sim$df_common, sim_options = df_sim$sim_options,
                 cond_err = 1, nerrs = NERRS, draw_mlhs = 1, sim_type = "welfare")

  # mean over draws for each camper and policy; average loss over the single-campsite policies (the last policy closes everything)
  per_camper <- t(sapply(w, colMeans, na.rm = TRUE))                    # campers x policies
  out <- tibble(id = as.character(unique(dat$id)),
                loss = -rowMeans(per_camper[, seq_len(J), drop = FALSE]))
  saveRDS(out, file)
  out
}

mdcev_losses <- function(yr) {

  fit <- mdcev_model_list[[as.character(yr)]]
  sd  <- fit$stan_data                                   # trips, prices and income in the order rmdcev uses
  x <- sd$quant_j;  p <- sd$price_j;  y <- as.numeric(sd$income);  N <- sd$I;  J <- sd$J

  sites <- as.character(fit$parms_info$alt_names)       # campsites in the model's order
  d     <- filter(camp, year == yr)

  # camper x campsite tables, in the same order as p. rmdcev keeps campers in the order of the data: try that order first
  ids   <- as.character(unique(final_data[final_data$year == yr, ]$id))
  put   <- function(v) { m <- matrix(NA_real_, N, J)
                         m[cbind(match(d$id, ids), match(d$parkname, sites))] <- v; m }
  if (max(abs(put(d$price) - p), na.rm = TRUE) > 1e-6) {  # otherwise try sorted ids
    ids <- sort(unique(as.character(d$id)))
  }
  price <- put(d$price)
  miles <- put(d$miles)
  extra <- put(d$leftover)
  rate  <- put(d$rate_per_mile)[, 1]
  stopifnot("price table does not match the model's price table" = max(abs(price - p), na.rm = TRUE) < 1e-6)

  # estimated parameters (point estimates), as in your fast script
  pt <- unlist(fit$stan_fit$par[c("psi", "gamma", "alpha", "scale")])
  names(pt) <- c(paste0("psi[1,", seq_len(J - 1), "]"), paste0("gamma[1,", seq_len(J), "]"), "alpha[1,1]", "scale[1]")
  lpsi  <- c(0, pt[grepl("^psi\\[", names(pt))])
  g     <- pt[grepl("^gamma\\[", names(pt))]
  a1    <- pt[grepl("^alpha\\[", names(pt))][1]
  scale <- pt[grepl("^scale\\[", names(pt))][1]

  # four price tables. The MDCEV needs every price to be positive (it divides by price), so a counterfactual price below $1
  # (possible when the leftover is negative at very short distances) is set to $1; the message says how many.
  floor_price <- function(m, label) {
    n <- sum(m < 1)
    if (n > 0) message("  ", yr, ", ", label, ": ", n, " of ", length(m), " prices below $1 were set to $1")
    pmax(m, 1)

  # campers who would spend more than 95% of income on trips in ANY of the four price tables are left out of all four

    out[affordable] <- rowMeans(md_welfare_person_cond(CD, xs, ps, ys, g, a1))
    out
  }
